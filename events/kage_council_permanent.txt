# ============================================================
# УКРЕПЛЕНИЕ СОВЕТА — предложение раз в 2 года
# ============================================================
country_event = {
    id = kage_council_permanent.3
    title = kage_council_permanent.3.t
    desc = kage_council_permanent.3.d
    picture = GFX_evt_generic_asia
    is_triggered_only = no
    fire_only_once = no
    major = yes

    trigger = {
        has_country_flag = in_kage_council
        has_country_flag = kage_council_active
        NOT = { has_country_flag = kage_council_strong }
        NOT = { has_country_flag = kage_council_disbanded }
        date > 1939.1.1
    }
    mean_time_to_happen = {
        months = 24
    }

    option = {
        name = kage_council_permanent.3.a # Укрепить Совет
        add_political_power = 100
        set_country_flag = voted_strong_council

        # Если большинство проголосовало — Совет становится сильным
        if = {
            limit = {
                any_country = {
                    has_country_flag = voted_strong_council
                    count = 3
                }
            }
            every_country = {
                limit = { has_country_flag = in_kage_council }
                remove_ideas = kage_council_active_idea
                add_ideas = kage_council_strong_idea
                set_country_flag = kage_council_strong
                clr_country_flag = voted_strong_council
            }
        }
    }

    option = {
        name = kage_council_permanent.3.b # Оставить как есть
        add_stability = 0.03
    }
}
