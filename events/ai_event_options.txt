# ---- Пример: Шиноби Альянс ----
country_event = {
    id = shinobi_alliance.1
    title = shinobi_alliance.1.t
    desc = shinobi_alliance.1.d
    picture = GFX_evt_generic_war
    is_triggered_only = no
    fire_only_once = yes
    major = yes

    trigger = {
        OR = { tag = KNO tag = SUN tag = KIR tag = NKU tag = IWA }
        AKA = { exists = yes has_country_flag = captured_kurama }
    }
    mean_time_to_happen = { days = 30 }

    option = {
        name = shinobi_alliance.1.a # Создать Альянс
        ai_chance = {
            base = 90
            modifier = {
                add = 10
                has_country_flag = pain_invasion_survived
            }
        }
        add_political_power = 100
        add_stability = 0.10
        add_war_support = 0.15
        set_country_flag = shinobi_alliance_founder
        if = {
            limit = { NOT = { is_in_faction = yes } }
            create_faction = "Альянс Шиноби"
        }
        every_country = {
            limit = {
                OR = { tag = KNO tag = SUN tag = KIR tag = NKU tag = IWA }
                NOT = { tag = PREV }
            }
            country_event = { id = shinobi_alliance.2 days = 2 }
        }
    }

    option = {
        name = shinobi_alliance.1.b # В одиночку
        ai_chance = { base = 10 }
        add_war_support = 0.10
        add_stability = -0.05
        set_country_flag = shinobi_alliance_refused
    }
}
