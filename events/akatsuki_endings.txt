country_event = {
    id = akatsuki_ending.1
    title = akatsuki_ending.1.t
    desc = akatsuki_ending.1.d
    picture = GFX_evt_generic_asia
    is_triggered_only = yes
    major = yes

    option = {
        name = akatsuki_ending.1.a
        add_stability = 0.30
        add_political_power = 500
        add_ideas = AKA_god_emperor_idea
        set_country_flag = madara_god_victory

        # Все страны — под контролем Акацуки
        every_country = {
            limit = {
                NOT = { tag = AKA }
                is_major = yes

            country_event = { id = akatsuki_ending.1.target days = 2 }
        }
    }
}
